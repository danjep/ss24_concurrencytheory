\section*{Aufgabe \printsheet.1} \setcounter{section}{1}
Let $N=(S,T,W)$ be a Petri net with connectivity matrix $\C$ and $M_1,M_2 \in \N^{|S|}, \sigma = t_i \dots t_m \in T^*$ such that $M_1 [ \sigma \rangle M_2$. To prove: $M_2 = M_1 + \C \cdot \psi(\sigma)$ \\

\begin{proof}
From the definition of the connectivity matrix it follows that: 
\[\C = \left( W(t_1, -) - W(-,t_1) \dots W(t_{|T|}, -) - W(-, t_{|T|}) \right)\]
Thus the term $\C \cdot \psi(\sigma)$ results in
\[\C \cdot \psi(\sigma) = \left( W(t_i,-) - W(-,t_i) + \dots + W(t_m,-) - W(-,t_m) \right)\]
and therefore according to the definition of the firing relation:
\[M_1 [\sigma \rangle M_2 = M_1 + W(t_i,-) - W(-,t_i) + \dots + W(t_m,-) - W(-,t_m) = M_1 + \C \cdot \psi(\sigma)\]
Thus the proposition is proven. \end{proof}

\subsection*{(a)}
Let $(N,M_0,M_f)$ be a petri-net with $N = (S,T,W)$. We construct a VASS $A=(Q,\Delta,q_0,v_0)$ in which the final configuration $(q_f,v_f)$ is reachable from $(q_0,v_0)$ if and only if $M_f$ is reachable from $M_0$ in $N$.
\begin{itemize}
\item $Q = \{q\}$
\item $q_0 = q$
\item $v_0 = M_0 = \myvec{M_0 (s_1)& \dots & M_0(s_d)}^T$
\item $\Delta = \{\forall t_h \in T: (q,a^h,q)\}$ where $a^h = \myvec{a_1^h& \dots &a_d^h}^T \in \Z^d$ is defined as:
\[a_i^h = W(t_h,s_i) - W(s_i,t_h)\]
\end{itemize}
The initial state is defined as $(q,M_0)$ and the final state is defined as $(q_f,v_f) = (q,M_f)$.

\begin{proposition}
$(q,M_f)$ is reachable from $(q,M_0)$ ($(q,M_f) \in Reach((q,M_0))$) if and only if $M_f$ is reachable from $M_0$ in $N$ ($M_f \in Reach(M_0)$).
\end{proposition}

\begin{proof}
\begin{eqnarray*}
M_f \in Reach(M_0) &\Leftrightarrow & \exists \sigma = t_b,\dots ,t_m: M_0 [\sigma \rangle M_f \\
&\Leftrightarrow & M_f = M_0 + W(t_b,-) - W(-,t_b) + \dots + W(t_m,-) - W(-,t_m) \\
&\Leftrightarrow & M_f = M_0 + a^b + \dots + a^m \quad \mid \text{since }a_i^h = W(t_h,s_i) - W(s_i,t_h)\\
&\Leftrightarrow & v_f = v_0 + a^b + \dots + a^m \\
&\Leftrightarrow & (q_0,v_0) \in Reach(q_f,v_f) \quad \mid \text{since the state never changes} 
\end{eqnarray*}
\end{proof}

\subsection*{(b)}
Let $A=(Q,\Delta,q_0,v_0)$ be a VASS of dimension $d \in \N$. We construct a petri net $(N,M_0,M_f)$ with $(q_f,v_f) \in Reach((q_0,v_0))$ if and only if $M_f \in Reach(M_0)$ as follows: \\

Let $N$ be $(S,T,W)$ with:
\begin{itemize}
\item $S = \{s_1, \dots s_d, s_q\}$
\item $T = \{t_\delta \mid \forall \delta \in \Delta\}$
\end{itemize}

The weight function $W:(S\times T) \cup (T\times S) \rightarrow \N$ is defined for any $\delta = (q_k,a,q_l) \in \Delta:$ \\ \[W(s_i,t_\delta)= \begin{cases}
 -a_i & \text{if } a_i < 0 \\
 0 & \text{otherwise}
 \end{cases},
 \quad W(t_\delta,s_i) = \begin{cases}
 a_i & \text{if } a_i >0 \\
 0 & \text{otherwise}
 \end{cases}\] and \\
 \[ W(t_\delta, s_q) =\begin{cases} (l-k)  & \text{if } l > k \\ 0 & \text{otherwise} \end{cases},
\quad W(s_q,t_\delta) = \begin{cases} (k-l)& \text{if } k>l\\ 0& \text{otherwise} \end{cases}\]


The initial marking is $M_0 = \myvec{v_{0_1} & \dots & v_{0_d} & 0}^T$ and the final marking $M_f = \myvec{v_{f_1} & \dots & v_{f_d} & m}^T$ where $q_m = q_f$.
\begin{proposition}
$(q,M_f)$ is reachable from $(q,M_0)$ ($(q,M_f) \in Reach((q,M_0))$) if and only if $M_f$ is reachable from $M_0$ in $N$ ($M_f \in Reach(M_0)$).
\end{proposition}


\begin{proof}
Let $\delta = (q_i,a,q_j)$ be a transition in the VASS with $v$ being the counter assignment before the transition and $(v+a)$ the assignment after the transition. In the formed petri net, the transition $t_\delta$ should have the same effect as $\delta$ in the VASS, meaning, the marking before the transition should be $M_i = \myvec{v_1 & \dots & v_d & i}^T$ and after the transition it should be $M_j = \myvec{v_1+a_1 & \dots v_d+a_d & j}^T$.

First we prove, that $M_j(s_q) = j$ if $M_i(s_q) = i$. For this we need to inspect two cases: \\

1. If $i < j$, then $j-i > 0$. Therefore there exists a weight $W(t_\delta,s_q) = j-i$. Therefore, if $t_\delta$ is triggered, it adds $(j-i)$ tokens to $s_q$. Since $i + (j - i) = j$, $s_q$ has $j$ tokens in marking $M_j$.

2. If $i > j$, then $j-i < 0$. Therefore there exists a weight $W(t_\delta,s_q) = i-j$. Therefore, if $t_\delta$ is triggered, it takes $(i-j)$ tokens from $s_q$. Since $i - (i - j) = i-i+j = j$, $s_q$ has $j$ tokens in marking $M_j$.

Now, we show the equivalence of the transitions at all other places. \\
\begin{eqnarray*}
&& \delta = (q_i,a,q_j) \in \Delta \land v \in \N^d \text{ assignment before transition} \\ 
&\Leftrightarrow & t_\delta \in T \land \forall a_i > 0: W(t_\delta,s_i) = a_i \land \forall a_i < 0: W(s_i,t_\delta) = -a_i \\
&\Leftrightarrow & M_j(s_1 ,\dots,s_d) = M_i(s_1,\dots,s_d) + W(t_\delta,s_1,\dots,s_d) - W(s_1,\dots,s_d,t_\delta) \\
&\Leftrightarrow & \forall k \in \{1,...,d\}: M_j(s_k) = M_i(s_k) + W(t_\delta,s_k) - W(s_k,t_\delta) = v_k + a_k \\
&\Leftrightarrow & M_j = \myvec{v_1+a_1 & \dots & v_d+a_d & j} 
\end{eqnarray*}

Since this shows that any transition $\delta$ can be replicated by $t_\delta$ in the petri net, if there exists a sequence of transitions $\delta_1,\dots, \delta_m$ with $(q_0,v_0) \overset{\delta_1}{\rightarrow} \dots \overset{\delta_m}{\rightarrow} (q_f,v_f)$, there has to be a sequence $\sigma = t_{\delta_1} \dots t_{\delta_m}$ with $M_0 [\sigma \rangle M_f$ and the other way around.
\end{proof}
