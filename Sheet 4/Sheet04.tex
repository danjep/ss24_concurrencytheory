\section*{Exercise \printsheet.3}
\subsection*{(a)}
Definition: $\sim_A \subseteq \Sigma^* \times \Sigma^*: v \sim_A w $ if $\forall q,q' \in Q: q \overset{v}{\to} q' \Leftrightarrow q \overset{w}{\to} q'$ \\

\begin{proposition}
$\sim_A$ is an equivalence relation.
\end{proposition}
\begin{proof} The attributes of an equivalence relation are reflexivity, symmetry and transitivity. \\
\subsubsection*{(i) Reflexivity}
\[\forall q,q' \in Q: q \overset{v}{\rightarrow} q' \Leftrightarrow q \overset{v}{\to} q' \Rightarrow v \sim_A v\]

\subsubsection*{(ii) Symmetry}
\begin{align*}
v \sim_A w &\Rightarrow \forall q,q' \in Q: q \overset{v}{\to} q' \Leftrightarrow  q \overset{w}{\to} q' \\
&\Leftrightarrow \forall q,q' \in Q: q \overset{w}{\to} q' \Leftrightarrow  q \overset{v}{\to} q' \\
& \Rightarrow w \sim_A v
\end{align*}

\subsubsection*{(iii) Transitivity}
\begin{align*}
u \sim_A v \land v \sim_a w &\Rightarrow \forall q,q' \in Q: q \overset{u}{\to} q \Leftrightarrow  q \overset{v}{\to} q'  \land \forall q,q' \in Q: q \overset{v}{\to} q' \Leftrightarrow  q \overset{w}{\to} q' \\
& \Rightarrow \forall q,q' \in Q: q \overset{v}{\to} q' \Leftrightarrow q \overset{v}{\to} q' \Leftrightarrow  q \overset{w}{\to} q' \\
& \Rightarrow \forall q,q' \in Q: q \overset{u}{\to} q' \Leftrightarrow  q \overset{w}{\to} q' \\
& \Rightarrow u \sim_A w
\end{align*}
\end{proof}

\begin{proposition}
$\sim_A$ has finitely many equivalence classes.
\end{proposition}
\begin{proof}
Assume, there are infinitely many equivalence classes $[v_1]_{\sim_A},[v_2]_{\sim_A},\dots$, then there must be infinitely many different states $q_i \in Q$. The reason is that each equivalence class $[v]_{\sim_A}$ is defined essentially by combinations $(q_i,q_j),(q_k,q_l),\dots \subseteq Q\times Q$ of states for which $q_i \overset{v}{\to} q_j$ holds. Therefore there must be infinitely many combinations of states to define these classes. \\  
Since A is finite and therefore has finitely many states, the number of different combinations of states is limited. Since each equivalence class is defined through having a unique set of combinations of states, the number of equivalence classes must be limited too.
\end{proof}

\begin{proposition} For all $w \in \Sigma^*: [w]_{\sim_A}$ is a regular language. \end{proposition}
\begin{proof}
First, we show that $L_{q,q'} = \{w \in \Sigma^*: q \overset{w}{\to} q'\}$ is a regular language. 
%We use the following properties of regular languages:

%\end{itemize}
%The language of a single character $v \in \Sigma$ is regular. Using (1), we know that for every $w \in \Sigma^*$ with $q \overset{w}{\to} q'$, $\{w\}$ is regular. \\
%Using (2), we know that $L_{q,q'} = \bigcup_{w \in \Sigma^*: q \overset{w}{\to} q'}$ must be regular. \\

For each pair of states $q,q' \in Q$ we can define a NFA $A' = (Q,\Sigma,q,\delta,\{q'\})$ that accepts $L_{q,q'}$.

We can now use that regular languages are closed under union to prove that $[w]_{\sim_A}$ must be regular, since
\[[w]_{\sim_A} = \bigcup_{q \in Q: q \overset{w}{\to} q'} L_{q,q'}\]
\end{proof}
